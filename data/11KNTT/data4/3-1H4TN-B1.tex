\begin{ex}%[1C4K1-6]
	Cho hình tứ diện $A B C D$. Gọi $I$ là trung điểm của cạnh $C D$ và $M, N$ lần lượt là trọng tâm của các tam giác $B C D, A C D$. Gọi $G$ là giao điểm của $A M$ và $B N$. Tính $\dfrac{G A}{GM}$.\\
	\shortans[3]{$3$}
	\loigiai{
				\immini
			{
				Do $M, N$ lần lượt là trọng tâm của các tam giác $B C D, A C D$ nên $\dfrac{I M}{M B}=\dfrac{I N}{N A}=\dfrac{1}{2}$.\\
				Xét tam giác $A B I$, ta có: $\dfrac{I M}{M B}=\dfrac{I N}{N A}$ nên theo định lí Thalès đảo, suy ra $M N \parallel A B$.\\
				Tam giác $GMN$ đồng dạng với tam giác $GAB$ nên ta có: $\dfrac{G M}{G A}=\dfrac{G N}{G B}=\dfrac{M N}{A B}=\dfrac{I M}{I B}=\dfrac{1}{3}$. Suy ra $\dfrac{G A}{GM}=3$.
				}{
				\begin{tikzpicture}[line join=round,line cap=round,line width=.6pt,font=\footnotesize,scale=1]
					\coordinate[label=left:$B$] (B) at (0,0);

					\coordinate[label=above left:$A$] (A) at (1.8,3.5);

					\coordinate (E) at ($(B)!0.5!(C)$);

					\path[name path=bn] (B)--(N); 
					\path[name intersections={of= am and bn,by=G}];

\begin{ex}%[1H2K1-3]%Câu 6.
	Cho hình chóp $S.ABCD$ có đáy là hình thang $ABCD$ với $AD\parallel BC$ và $AD=2BC$. Gọi $M$ là điểm trên cạnh $SD$ thỏa mãn $SM=\dfrac{1}{3}SD$. Mặt phẳng $(ABM)$ cắt cạnh bên $SC$ tại điểm $N$. Tính tỉ số $\dfrac{SN}{SC}$.\\
	\shortans[3]{$0,5$}
	\loigiai{
		\begin{center}
			\begin{tikzpicture}[scale=1, font=\footnotesize,line join=round, line cap=round, >=stealth]
				\def\h{3}
				\path (0,0)coordinate(A)--(4,0)coordinate(D)--(1.2,-1.5)coordinate(B)--(3.2,-1.5)coordinate(C)--($(A)+(2,\h)$)coordinate(S);
				\coordinate (M) at ($(S)!1/3!(D)$);
				\coordinate (F) at (intersection of A--B and C--D);
				\coordinate (N) at (intersection of F--M and S--C);
				\coordinate (e) at ($(M)+(C)-(N)$);
				\coordinate (E) at (intersection of C--e and S--D);
				\draw (S)--(A)--(B)--cycle (S)--(C)--(D)--cycle (B)--(S)--(C) (B)--(F)--(C) (F)--(M) (B)--(N) (C)--(E);
				\draw[dashed] (A)--(D) (C)--(B) (A)--(M)--(B);
				\foreach \x/\g in {S/90,A/150,D/30,C/-60,B/-120,F/-90,M/60,N/-10,E/50}\fill (\x) circle (1pt)+(\g:0.25)node{$\x$};
			\end{tikzpicture}
		\end{center}
		Gọi $F$ là giao điểm của $AB$ và $CD$. Nối $F$ với $M$, $FM$ cắt $SC$ tại điểm $N$. Khi đó $N$ là giao điểm của $(ABM)$ và $SC$.\\
		Theo giả thiết, ta chứng minh được $C$ là trung điểm $DF$.\\
		Trong mặt phẳng $(SCD)$ kẻ $CE$ song song $NM$ ($E$ thuộc $SD$).\\
		Do $C$ là trung điểm $DF$ nên suy ra $E$ là trung điểm $MD$.\\
		Khi đó, ta có $SM=ME=ED$ và $M$ là trung điểm $SE$.\\
		Do $MN\parallel CE$ và $M$ là trung điểm $SE$ nên $MN$ là đường trung bình của tam giác $SCE$.\\
		Từ đó suy ra $N$ là trung điểm $SC$ và $\dfrac{SN}{SC}=\dfrac{1}{2}$.}
\end{ex}

\begin{ex}%[Thi giữa HKI, THPT Nam Trực, Nam Định, 2018-2019]%[Đỗ Đường Hiếu, 12-EX-3-2019]%[1H2K1-3]%
	Cho hình chóp $S.ABCD$ có đáy là hình bình hành tâm $O$. Gọi $M$, $N$, $P$ lần lượt là trung điểm của $SB$, $SD$ và $OC$. Gọi giao điểm của $(MNP)$ với $SA$ là $K$. Tính tỉ số $\dfrac{KA}{KS}$.\\
	\shortans[3]{$3$}
	\loigiai{
		\begin{center}
			\begin{tikzpicture}[scale=1, font=\footnotesize,line join=round, line cap=round,>=stealth]
				\tkzDefPoints{-3/-2/B,0/0/A,5/0/D,0/4/S}
				\tkzDefPointBy[translation=from A to D](B)\tkzGetPoint{C}
				\tkzDefMidPoint(S,B)\tkzGetPoint{M}
				\tkzDefMidPoint(S,D)\tkzGetPoint{N}
				\tkzInterLL(A,C)(B,D)\tkzGetPoint{O}
				\tkzDefMidPoint(O,C)\tkzGetPoint{P}
				\tkzInterLL(S,O)(M,N)\tkzGetPoint{I}
				\tkzInterLL(P,I)(S,A)\tkzGetPoint{K}
				\tkzDefMidPoint(O,A)\tkzGetPoint{J}
				\tkzDrawSegments(B,C C,D S,B S,D S,C)
				\tkzDrawSegments[dashed](A,B A,D S,A M,N M,P N,P A,C B,D S,O P,K I,J)
				\tkzDrawPoints[fill=black](A,B,C,D,S,O,M,N,P,I,K,J)
				\tkzLabelPoints[left](B,M,K)
				\tkzLabelPoints[right](D,C,N,J)
				\tkzLabelPoints[above](S)
				\tkzLabelPoints[above right](A)
				\tkzLabelPoints[below](O,P)
				\tkzLabelPoints[below left](I)
			\end{tikzpicture}
		\end{center}
		Gọi $J$ là trung điểm $OA$, ta có $IJ\parallel SA$. Khi đó
		$\dfrac{IJ}{KA}=\dfrac{PJ}{PA}=\dfrac{2}{3}\Rightarrow KA=\dfrac{3}{2}IJ$.\qquad $(1)$\\
		Gọi $I=MN\cap SO$, $K=PI\cap SA$. Khi đó $K=SA\cap (MNP)$ và $I$ là trung điểm của $SO$, $SA=2IJ$.\qquad $(2)$\\
		Từ $(1)$ và $(2)$ suy ra $\dfrac{SA}{KA}=\dfrac{4}{3}\Rightarrow \dfrac{KS}{KA}=\dfrac{1}{3} \Rightarrow \dfrac{KA}{KS}=3$.
	}
\end{ex}
