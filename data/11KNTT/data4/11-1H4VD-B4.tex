\begin{dang}{Chứng minh đường thẳng song song với mặt phẳng}
	Để chứng minh $a$ song song $(P)$, ta thường sử dụng một trong hai cách sau
	\begin{enumerate}[\iconMT]
		\item \indam{Cách 1: } (\textit{Đã xét ở bài học trước}) Ta cần chứng tỏ các ý sau:
		\begin{itemize}
			\item [$\bullet$] $a$ không nằm trên $(P)$;
			\item [$\bullet$] $a$ song song với một đường thẳng $b$ nằm trong $(P)$. Suy ra $a\parallel (P)$ hay  \boxmini{$\heva{&a\not\subset (P)\\&a\parallel b \\&b\subset (P)\\}\Rightarrow a\parallel (P)$}
		\end{itemize}
		\item \indam{Cách 2:} Ta chứng minh đường thẳng $a$ nằm trong mặt
		phẳng $(Q)$ và $(Q)\parallel (P)$ thì $a \parallel (P)$.
	\end{enumerate}
\end{dang}

\begin{bt}%[Trần Thị Thu Hằng]%[1H2K4]
	Cho hình chóp $S.ABCD$, đáy $ABCD$ là hình thang có $AB \parallel CD$ và $AB=2CD$, $I$ là giao điểm của $AC$ và $BD$. Gọi $M$ là trung điểm của $SD$, $E$ là trung điểm đoạn $CM$ và $G$ là điểm đối xứng của $E$ qua $M$, $SE$ cắt $CD$ tại $K$. Chứng minh $(IKE) \parallel (ADG)$.
	\loigiai{
		\immini{Do $CE=ME=MG$ nên \begin{align}\label{2.1}
				CE=\dfrac{1}{3} CG. \tag{1}
			\end{align}
			Mặt khác $$\begin{cases} \widehat{BAI}&=\widehat{DCI},\text{ (so le trong)}, \\ \widehat{AIB}&=\widehat{CID},\text{ (đối đỉnh)}. \end{cases}$$
			Do đó $\triangle ABI \backsim \triangle CDI$, (g-g).
			Khi đó \begin{align}\label{2.2}
				\dfrac{CI}{IA}=\dfrac{CD}{AB}=\dfrac{1}{2}\hspace{5pt}\text{hay}\hspace{5pt} \dfrac{CI}{CA}=\dfrac{1}{3}.\tag{2}
			\end{align}
			Từ \eqref{2.1} và \eqref{2.2} suy ra 
			\begin{align}\label{*}
				EI \parallel GA.\tag{$\star$}
		\end{align}}{

			$ \Rightarrow (G_1G_2G_3)\parallel (BCD) $.\\
			}

\begin{bt}%[1H2K4-2]
	Cho hình chóp $SABC$ có $G$ là trọng tâm tam giác $ABC$. Trên đoạn $SA$ lấy hai điểm $M$, $N$ sao cho $SM=MN=NA$.
	\begin{listEX}[1]
		\item Chứng minh rằng $GM\parallel (SBC)$. 
		\item Gọi $D$ là điểm đối xứng với $A$ qua $G$. Chứng minh rằng $(MCD)\parallel (NBG)$.
		\end{listEX}

		\begin{enumerate}

			\item Từ giả thiết ta suy ra $G,N$ lần lượt là trung điểm của $AD$ và $AM$. Do đó $NG \parallel MD \quad (1)$ \hfill $(1)$\\
			Từ giác $BDCG$ có $E$ là trung điểm của hai đường chéo nên đó là hình bình hành. Suy ra $BG\parallel CD$ \hfill $(2)$\\

		\end{enumerate}
	}

\begin{bt}
\immini{Cho hình hộp $ABCD.A'B'C'D'$  . Một mặt phẳng song song với mặt đáy $(ABCD)$  của hình hộp và cắt các cạnh $AA', BB', CC', DD'$  lần lượt tại $M, N, M', N'$ . Chứng minh rằng $ABCD.MNM'N'$  là hình hộp.}{
	\begin{tikzpicture}[scale=0.55, font=\footnotesize, line join=round, line cap=round]
		\foreach \x\y\t in {0/0/A,-2/-1.5/B,3.9/0/D,-0.5/3.5/A'}
		\coordinate (\t) at (\x,\y);
		\coordinate (C) at ($(B)+(D)-(A)$);
		\coordinate (B') at ($(A')+(B)-(A)$);
		\coordinate (C') at ($(B')+(C)-(B)$);
		\coordinate (D') at ($(C')+(D)-(C)$);
		\draw (A')--(B')--(B)--(C)--(C');
		\draw (A')--(D')--(D);
		\draw (D')--(C') (C)--(D);
		\draw (B')--(C') (D)--(D');
		\draw[dashed] (A)--(B) (A')--(A)--(D);
		\foreach \t/\g in {A/180,B/180,C/0,D/0,A'/180,B'/180,C'/0,D'/0} \draw (\t) node[shift={(\g:10pt)}]{$\t$};
	\end{tikzpicture}
}
	\loigiai{
		Ta có mặt phẳng $(MNM'N') \parallel (ABCD)$ (theo giả thiết).\\
		Mặt phẳng $(ABB'A')$ cắt hai mặt phẳng song song $(ABCD)$ và $(MNM'N')$ lần lượt theo hai giao tuyến là $AB$ và $MN$. Áp dụng tính chất giao tuyến của hai mặt phẳng song song, ta suy ra $MN \parallel AB$.\\
		Tương tự, xét các mặt bên còn lại của hình hộp, ta chứng minh được: $NM' \parallel BC$, $M'N' \parallel CD$ và $N'M \parallel DA$.\\
		Mặt khác, vì đáy $ABCD$ là hình bình hành nên $AB \parallel CD$ và $AD \parallel BC$.\\
		Từ đó suy ra $MN \parallel M'N'$ (cùng song song với hai cạnh đáy) và $N'M \parallel NM'$. Do đó, tứ giác $MNM'N'$ là hình bình hành.\\
		Xét khối đa diện $ABCD.MNM'N'$:
		\begin{itemize}
			\item Có hai đáy $(ABCD)$ và $(MNM'N')$ là hai hình bình hành nằm trong hai mặt phẳng song song.
			\item Các cạnh bên $AM, BN, CM', DN'$ đôi một song song với nhau (do chúng là các đoạn thẳng nằm trên các cạnh bên song song của hình hộp gốc $ABCD.A'B'C'D'$).
		\end{itemize}
		Vậy đa diện $ABCD.MNM'N'$ là một hình hộp.
	}
\end{bt}\cham{14}
