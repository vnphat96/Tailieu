\begin{dang}{Các quan hệ cơ bản}
	\begin{itemize}
		\item [\ding{172}] Chứng minh điểm $A$ thuộc $(\alpha)$: %Ta chứng tỏ điểm $A$ thuộc đường thẳng $\Delta$ nằm trong $\alpha$, nghĩa là
		$\heva{&A \in \Delta \\& \Delta \subset (\alpha)} \Rightarrow A \in (\alpha).$
		\item [\ding{173}] Chứng minh đường thẳng $d$ nằm trong $(\alpha)$: %Ta chứng tỏ $d$ có hai điểm phân biệt cùng thuộc $(\alpha)$, nghĩa là
		$\heva{&A \in (\alpha), B \in (\alpha)\\& A,\,B \in d} \Rightarrow d \subset (\alpha).$
		\item [\ding{174}] Chứng minh $A$ là điểm chung của hai mặt phẳng $(\alpha)$ và $(\beta)$:%Ta thường sử dụng một trong hai cách sau
		$$\heva{&A \in (\alpha) \\& A \in (\beta)} \Rightarrow A \in (\alpha) \cap  (\beta).
		\quad \text{ hoặc }\quad
		\heva{&d \subset (\alpha), \Delta \subset (\beta)\\& d \cap \Delta = A} \Rightarrow A \in (\alpha) \cap  (\beta).$$
	\end{itemize}
\end{dang}

	\loigiai{
		\textbf{Nhắc lại kiến thức:}
		\begin{itemize}
			\item Trọng tâm của tam giác chia mỗi đường trung tuyến thành hai phần, phần nối từ đỉnh đến trọng tâm bằng $\dfrac{2}{3}$ độ dài đường trung tuyến đó.
			\item Định lý Thales đảo: Nếu một đường thẳng cắt hai cạnh của một tam giác và định ra trên hai cạnh này những đoạn thẳng tương ứng tỉ lệ thì đường thẳng đó song song với cạnh còn lại của tam giác.
		\end{itemize}
		
		\textbf{Áp dụng:}
		\begin{enumerate}[a)]
			\item  Chứng minh các điểm $M, N$ thuộc mặt phẳng $(ABI)$:
		\begin{itemize}
		\item Trong tam giác $BCD$, vì $I$ là trung điểm của $CD$ nên $BI$ là đường trung tuyến.
		Theo giả thiết, $M$ là trọng tâm của $\Delta BCD$ nên $M$ nằm trên đường trung tuyến $BI$.
		Mà $BI \subset (ABI)$, do đó $M \in (ABI)$.
		\item Tương tự, trong tam giác $CDA$, $AI$ là đường trung tuyến.
		Vì $N$ là trọng tâm của $\Delta CDA$ nên $N$ nằm trên đường trung tuyến $AI$.
		Mà $AI \subset (ABI)$, do đó $N \in (ABI)$.
	\end{itemize}
Vậy hai điểm $M$ và $N$ đều thuộc mặt phẳng $(ABI)$.
		\item Chứng minh $\dfrac{GM}{GA} = \dfrac{GN}{GB} = \dfrac{1}{3}$:
		Theo chứng minh ở câu a), ta có $M \in BI$ và $N \in AI$, nên hai đường thẳng $AM$ và $BN$ cùng nằm trong mặt phẳng $(ABI)$. Gọi $G$ là giao điểm của chúng.
	\begin{itemize}
	\item Vì $M$ là trọng tâm $\Delta BCD$ nên ta có tỉ số: $\dfrac{IM}{IB} = \dfrac{1}{3}$.
	\item Vì $N$ là trọng tâm $\Delta CDA$ nên ta có tỉ số: $\dfrac{IN}{IA} = \dfrac{1}{3}$.
\end{itemize}
Xét tam giác $IAB$, ta có:
$$ \dfrac{IM}{IB} = \dfrac{IN}{IA} = \dfrac{1}{3}. $$
Theo định lý Thales đảo, ta suy ra $MN \parallel AB$.
Đồng thời, ta cũng có tỉ số $\dfrac{MN}{AB} = \dfrac{IN}{IA} = \dfrac{1}{3}$.

Xét tam giác $GAB$, vì $MN \parallel AB$, áp dụng hệ quả của định lý Thales, ta có:
$$ \dfrac{GM}{GA} = \dfrac{GN}{GB} = \dfrac{MN}{AB}. $$
Mà $\dfrac{MN}{AB} = \dfrac{1}{3}$ (chứng minh trên), suy ra:
$$ \dfrac{GM}{GA} = \dfrac{GN}{GB} = \dfrac{1}{3}. $$
(Điều phải chứng minh).
		\end{enumerate}
	}
\end{vd}
\begin{dang}{Xác định giao tuyến của hai mặt phẳng}
	Cho hai mặt phẳng $(\alpha)$ và $(\beta)$ cắt nhau. Để xác định giao tuyến của chúng, ta đi tìm hai điểm chung phân biệt. Cụ thể, ta thường gặp một trong ba trường hợp sau:
\begin{itemize}
		\item [\ding{172}] Hai mặt phẳng $(\alpha)$ và $(\beta)$ có sẵn hai điểm chung phân biệt: Khi đó giao tuyến là đường thẳng qua hai điểm chung đó.
		\item [\ding{173}] Hai mặt phẳng  $(\alpha)$ và $(\beta)$ thấy trước một điểm chung $A$:
		\begin{khung4}{}
			\begin{itemize}
				\item [$\bullet$] $A$ là điểm chung thứ nhất hay $A \in (\alpha) \cap (\beta)$.
				\item [$\bullet$] Ta tìm điểm chung thứ 2: Trong $ (\alpha)$ tìm một đường thẳng $d_1$, trong $ (\beta)$ tìm một đường thẳng $d_2$ sao cho chúng có thể cắt nhau (đồng phẳng).			Gọi $B = d_1 \cap d_2$, suy ra $B \in (\alpha) \cap (\beta)$. Vậy $AB=(\alpha) \cap (\beta)$.
		\end{itemize}
		\end{khung4}
		\item [\ding{174}] Hai mặt phẳng  $(\alpha)$ và $(\beta)$ chưa thấy điểm chung: Ta mở rộng mặt phẳng để tìm điểm chung tương tự như cách tìm điểm chung ở mục số \ding{173}.
	\end{itemize}
\end{dang}
