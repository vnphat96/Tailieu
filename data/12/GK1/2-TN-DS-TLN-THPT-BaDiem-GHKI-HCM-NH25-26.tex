			& s(0)=0 \\
			& s(2)=12 \\
			& s(4)=24 \\
			& {s}'(4)=0}
		\Leftrightarrow \heva{
			& d=0 \\
			& 8a+4b+2c+d=12 \\
			& 64a+16b+4c+d=24 \\
			& 48a+8b+c=0}
		\Leftrightarrow \heva{
			& a=-\dfrac{3}{4} \\
			& b=\dfrac{9}{2} \\
			& c=0 \\
			& d=0.}$\\
